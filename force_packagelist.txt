# Always removed, with no volume-key prompt.
#
# Two groups live here:
#   1. Google apps that a previous install silently kept because the per-app
#      volume prompt timed out or was answered "No".
#   2. Google apps that have a bundled FOSS replacement, so dropping them can
#      never leave a hole (dialer, contacts, calendar, clock, calculator,
#      messages, files).
#
# Accessibility pieces (TalkBack, Google TTS) and core services are never
# listed here.

# --- Kept by earlier installs ---
com.google.android.gm                      # Gmail
com.google.android.googlequicksearchbox    # Google Search / Google app
com.google.android.apps.maps               # Maps
com.google.android.apps.photos             # Photos
com.google.android.apps.weather            # Pixel Weather
com.google.android.apps.nbu.files          # Files by Google

# --- Replaced by bundled FOSS apps ---
com.google.android.dialer                  # -> Fossify Phone
com.google.android.contacts                # -> Fossify Contacts
com.google.android.calendar                # -> Fossify Calendar
com.google.android.deskclock               # -> Fossify Clock
com.google.android.calculator              # -> CalcYou
com.google.android.apps.messaging          # -> Fossify Messages

# --- WebView (CONDITIONAL, see below) ---
com.google.android.webview                 # Google System WebView -> bundled com.android.webview

# The WebView entry above is applied only when the module actually ships a
# replacement provider. customize.sh checks for
# system/priv-app/WebViewGoogle/WebViewGoogle.apk and confirms it carries the
# com.android.webview identity before masking anything. If the replacement is
# missing, Google's WebView is left installed, because removing it without a
# working provider breaks every embedded page and login screen.

# --- Pure Google extras, nothing depends on them ---
com.google.android.apps.wellbeing          # Digital Wellbeing
com.google.android.printservice.recommendation  # Google print service
